cmake_minimum_required(VERSION 3.19)
project(headbutt_binding LANGUAGES CXX)
find_package(Python REQUIRED COMPONENTS Interpreter Development.Module)
find_package(pybind11 2.13 CONFIG REQUIRED)
if(NOT IS_DIRECTORY "${FORK_SOURCE}/include" OR NOT EXISTS "${STS_CORE_LIBRARY}")
  message(FATAL_ERROR "Build the pinned fork's sts_core and supply FORK_SOURCE and STS_CORE_LIBRARY")
endif()
pybind11_add_module(sts_env "${FORK_SOURCE}/bindings/sts_env.cpp")
target_compile_features(sts_env PRIVATE cxx_std_17)
target_include_directories(sts_env PRIVATE "${FORK_SOURCE}/include" "${FORK_SOURCE}/json/single_include")
target_link_libraries(sts_env PRIVATE "${STS_CORE_LIBRARY}")
if(MSVC)
  target_compile_options(sts_env PRIVATE /openmp /bigobj /permissive- /Zc:__cplusplus)
  target_compile_definitions(sts_env PRIVATE NOMINMAX _CRT_SECURE_NO_WARNINGS)
else()
  find_package(OpenMP REQUIRED)
  target_link_libraries(sts_env PRIVATE OpenMP::OpenMP_CXX)
endif()

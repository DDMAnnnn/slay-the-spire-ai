cmake_minimum_required(VERSION 3.19)
project(heart_policy LANGUAGES CXX)
find_package(Python REQUIRED COMPONENTS Interpreter Development.Module)
find_package(pybind11 2.13 CONFIG REQUIRED)
find_package(OpenMP REQUIRED)
pybind11_add_module(heart_native policy.cpp)
target_compile_features(heart_native PRIVATE cxx_std_17)
target_include_directories(heart_native PRIVATE "${FORK_SOURCE}/include" "${FORK_SOURCE}/json/single_include")
target_link_libraries(heart_native PRIVATE "${STS_CORE_LIBRARY}" OpenMP::OpenMP_CXX)
if(MSVC)
  target_compile_options(heart_native PRIVATE /bigobj /permissive- /Zc:__cplusplus /W1)
  target_compile_definitions(heart_native PRIVATE NOMINMAX _CRT_SECURE_NO_WARNINGS)
endif()
